cmake_minimum_required(VERSION 3.20)
project(garuda_perception CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(GARUDA_BUILD_NODE  "Build the on-board pipeline app (needs the Hailo-15L SDK env)" OFF)
option(GARUDA_BUILD_TESTS "Build unit tests (native or cross)" ON)

find_library(TURBOJPEG_LIB turbojpeg REQUIRED)

add_library(garuda_perception_core STATIC)
target_include_directories(garuda_perception_core PUBLIC include third_party)
# -ffp-contract=off: aarch64 has FMA, and GCC contracts a multiply-add sequence like
# `a*b + c*d - e` into fused instructions by default; that rounds differently than the separate
# multiply/add the Python reference (tests/golden/*_ref.py) and the golden JSON assume, and can move
# an IoU value ~1 ulp across a `>= iou_thr` boundary. The native x86-64 baseline build has no FMA, so
# without this flag the golden tests would pass here and still fail on an aarch64 cross-build.
target_compile_options(garuda_perception_core PRIVATE -Wall -Wextra -ffp-contract=off)
target_sources(garuda_perception_core PRIVATE src/snapshot.cpp src/feed_server.cpp src/jpeg_encoder.cpp src/bytetrack.cpp src/lock.cpp src/lock_core.cpp src/yolo26_decode.cpp src/hailo_track_adapter.cpp)
target_link_libraries(garuda_perception_core PUBLIC pthread ${TURBOJPEG_LIB})
# The postprocess plugin (postprocess/) links this static library into a SHARED object; without PIC
# the link fails with relocation errors (R_AARCH64_ADR_PREL_PG_HI21 etc against non-PIC .o's). Harmless
# for the node/test executables that also link it.
set_target_properties(garuda_perception_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(GARUDA_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

if(GARUDA_BUILD_NODE)
  include(${CMAKE_CURRENT_SOURCE_DIR}/../../hosts/hailo15l/cmake/hailo15l.cmake)
  add_executable(garuda-perception src/node_main.cpp)
  target_compile_options(garuda-perception PRIVATE -Wall -Wextra)
  target_include_directories(garuda-perception PRIVATE ${GARUDA_HAILO_INCLUDE})
  target_link_libraries(garuda-perception PRIVATE garuda_perception_core hailo_analytics hailo_postprocess_tools medialib hailort pthread)
  install(TARGETS garuda-perception RUNTIME DESTINATION bin)
  add_subdirectory(postprocess)
endif()
