cmake_minimum_required(VERSION 3.20)
project(garuda_tracking LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GARUDA_TRACKING_TESTS "Build deterministic tracking tests" ON)
option(GARUDA_TRACKING_OPENCV "Build optional OpenCV camera motion plugin" OFF)
option(GARUDA_TRACKING_VISUAL_TARGET "Build experimental selected-target visual localizer" OFF)
add_library(garuda_tracking STATIC src/tracker.cpp src/vision.cpp src/target_owner.cpp src/search_region.cpp src/tiling.cpp)
target_include_directories(garuda_tracking PUBLIC include)
target_compile_options(garuda_tracking PRIVATE -Wall -Wextra -Wpedantic)
add_library(garuda_tracking_json INTERFACE)
target_include_directories(garuda_tracking_json INTERFACE ../garuda-perception/third_party)
target_link_libraries(garuda_tracking_json INTERFACE garuda_tracking)
add_library(garuda_capture_stream STATIC src/capture_stream.cpp src/source_pairing.cpp)
target_link_libraries(garuda_capture_stream PUBLIC garuda_tracking_json pthread)
target_compile_options(garuda_capture_stream PRIVATE -Wall -Wextra -Wpedantic)
if(GARUDA_TRACKING_OPENCV)
  if(CMAKE_CROSSCOMPILING)
    # The vendor OpenCVConfig imports absent, unrelated contrib static archives.
    # Resolve only these shared modules inside the configured SDK sysroot.
    find_path(GARUDA_OPENCV_INCLUDE opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
    set(OpenCV_INCLUDE_DIRS ${GARUDA_OPENCV_INCLUDE})
    set(OpenCV_LIBS "")
    foreach(component core imgproc features2d calib3d)
      find_library(GARUDA_OPENCV_${component} NAMES opencv_${component} REQUIRED)
      list(APPEND OpenCV_LIBS ${GARUDA_OPENCV_${component}})
    endforeach()
  else()
    find_package(OpenCV REQUIRED COMPONENTS core imgproc features2d calib3d)
  endif()
  add_library(garuda_camera_motion STATIC src/camera_motion.cpp)
  target_include_directories(garuda_camera_motion PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(garuda_camera_motion PUBLIC garuda_tracking ${OpenCV_LIBS})
  target_compile_options(garuda_camera_motion PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(garuda-motion-probe src/camera_motion_probe.cpp)
  target_include_directories(garuda-motion-probe PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(garuda-motion-probe PRIVATE garuda_camera_motion garuda_tracking_json)
endif()
add_executable(garuda-track-replay src/replay.cpp)
target_link_libraries(garuda-track-replay PRIVATE garuda_tracking_json)
if(GARUDA_TRACKING_VISUAL_TARGET)
  if(CMAKE_CROSSCOMPILING)
    find_path(GARUDA_VISUAL_INCLUDE opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
    find_library(GARUDA_VISUAL_CORE NAMES opencv_core REQUIRED)
    find_library(GARUDA_VISUAL_IMGPROC NAMES opencv_imgproc REQUIRED)
    set(GARUDA_VISUAL_LIBS ${GARUDA_VISUAL_CORE} ${GARUDA_VISUAL_IMGPROC})
  else()
    find_package(OpenCV REQUIRED COMPONENTS core imgproc)
    set(GARUDA_VISUAL_INCLUDE ${OpenCV_INCLUDE_DIRS})
    set(GARUDA_VISUAL_LIBS ${OpenCV_LIBS})
  endif()
  add_library(garuda_visual_target STATIC src/visual_target.cpp)
  target_include_directories(garuda_visual_target PRIVATE ${GARUDA_VISUAL_INCLUDE})
  target_link_libraries(garuda_visual_target PUBLIC garuda_tracking ${GARUDA_VISUAL_LIBS})
  target_compile_options(garuda_visual_target PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(garuda-visual-replay ../../bench/hailo_reference/visual_target_replay.cpp)
  target_include_directories(garuda-visual-replay PRIVATE ${GARUDA_VISUAL_INCLUDE})
  target_link_libraries(garuda-visual-replay PRIVATE garuda_visual_target garuda_tracking_json)
endif()
if(GARUDA_TRACKING_TESTS)
  enable_testing()
  if(GARUDA_TRACKING_VISUAL_TARGET)
    add_executable(test_visual_target tests/test_visual_target.cpp)
    target_include_directories(test_visual_target PRIVATE ${GARUDA_VISUAL_INCLUDE})
    target_link_libraries(test_visual_target PRIVATE garuda_visual_target garuda_tracking_json)
    add_test(NAME selected_visual_target COMMAND test_visual_target)
  endif()
  add_executable(test_motion_samples tests/test_motion_samples.cpp)
  target_link_libraries(test_motion_samples PRIVATE garuda_tracking)
  add_test(NAME motion_sample_parity COMMAND test_motion_samples)
  add_executable(test_fresh_tracking tests/test_tracking.cpp)
  target_link_libraries(test_fresh_tracking PRIVATE garuda_tracking_json)
  add_test(NAME fresh_tracking COMMAND test_fresh_tracking)
  add_executable(test_target_owner tests/test_target_owner.cpp)
  target_link_libraries(test_target_owner PRIVATE garuda_tracking_json)
  add_test(NAME learned_target_owner COMMAND test_target_owner)
  add_executable(test_search_region tests/test_search_region.cpp)
  target_link_libraries(test_search_region PRIVATE garuda_tracking)
  add_test(NAME target_search_region COMMAND test_search_region)
  add_executable(test_diagnostic_budget tests/test_diagnostic_budget.cpp)
  target_compile_options(test_diagnostic_budget PRIVATE -UNDEBUG)
  add_test(NAME h15l_diagnostic_budget COMMAND test_diagnostic_budget)
  add_executable(test_capture_stream tests/test_capture_stream.cpp)
  target_link_libraries(test_capture_stream PRIVATE garuda_capture_stream)
  add_test(NAME bounded_capture_stream COMMAND test_capture_stream)
  add_executable(test_source_pairing tests/test_source_pairing.cpp)
  target_link_libraries(test_source_pairing PRIVATE garuda_capture_stream)
  add_test(NAME exact_source_pairing COMMAND test_source_pairing)
  add_executable(test_motion_transform tests/test_motion_transform.cpp)
  target_link_libraries(test_motion_transform PRIVATE garuda_tracking_json)
  add_test(NAME camera_motion_transform COMMAND test_motion_transform)
  if(GARUDA_TRACKING_OPENCV)
    add_executable(test_camera_motion tests/test_camera_motion.cpp)
    target_include_directories(test_camera_motion PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(test_camera_motion PRIVATE garuda_camera_motion garuda_tracking_json)
    add_test(NAME camera_similarity_estimator COMMAND test_camera_motion)
  endif()
endif()

install(TARGETS garuda-track-replay RUNTIME DESTINATION bin)
