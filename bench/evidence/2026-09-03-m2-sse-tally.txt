Script (checked against the same sse600.txt), run as: python3 -c "<script below>"

import json
n=0
person_present=0
lock_id_counts={}
min_tracks=None
max_tracks=0
with open('sse600.txt','rb') as f:
    for raw in f:
        if not raw.startswith(b'data: '):
            continue
        n+=1
        obj=json.loads(raw[6:].strip())
        tracks=obj.get('tracks',[])
        labels=[t.get('label') for t in tracks]
        if 'person' in labels:
            person_present+=1
        lid=obj.get('lock',{}).get('id')
        lock_id_counts[lid]=lock_id_counts.get(lid,0)+1
        nt=len(tracks)
        if min_tracks is None or nt<min_tracks: min_tracks=nt
        if nt>max_tracks: max_tracks=nt
print('n=',n,'person_present_frames=',person_present,'lock_id_counts=',lock_id_counts,'min_tracks=',min_tracks,'max_tracks=',max_tracks)
