--- script source ---
import json, sys

path = sys.argv[1]
n_data = 0
n_tracks_nonempty = 0
n_coasting = 0
lock_ids = set()
first5 = []
last5 = []
ages = []

with open(path, "rb") as f:
    for raw in f:
        if not raw.startswith(b"data: "):
            continue
        n_data += 1
        line = raw[6:].strip()
        if len(first5) < 5:
            first5.append(line.decode("utf-8", "replace"))
        last5.append(line.decode("utf-8", "replace"))
        if len(last5) > 5:
            last5.pop(0)
        try:
            obj = json.loads(line)
        except Exception as e:
            print(f"PARSE ERROR on data line {n_data}: {e}")
            continue
        tracks = obj.get("tracks", [])
        if tracks:
            n_tracks_nonempty += 1
        lock = obj.get("lock", {})
        if lock.get("coasting") is True:
            n_coasting += 1
        lid = lock.get("id")
        if lid is not None:
            lock_ids.add(lid)
        age = obj.get("age_ms")
        if age is not None:
            ages.append(age)

print(f"total data: lines = {n_data}")
print(f"frames with non-empty tracks = {n_tracks_nonempty}")
print(f"frames with lock.coasting == true = {n_coasting}")
print(f"distinct lock.id values seen = {sorted(lock_ids)}")
if ages:
    ages_sorted = sorted(ages)
    print(f"age_ms count={len(ages_sorted)} min={ages_sorted[0]} max={ages_sorted[-1]}")

print("\n--- first 5 data lines ---")
for l in first5:
    print(l)
print("\n--- last 5 data lines ---")
for l in last5:
    print(l)
